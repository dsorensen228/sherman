cmake_minimum_required(VERSION 3.20)
option(SHERMAN_PICO "Build the diagnostic-only Pico 2 target" OFF)
option(SHERMAN_SANITIZERS "Host ASan and UBSan" ON)
if(SHERMAN_PICO)
  if(NOT DEFINED PICO_SDK_PATH)
    set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/../../.cache/pico/sdk")
  endif()
  set(PICO_BOARD pico2 CACHE STRING "W1 board")
  if(NOT PICO_BOARD STREQUAL "pico2")
    message(FATAL_ERROR "Only non-wireless Pico 2 W1 is supported")
  endif()
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${PICO_SDK_PATH}
                  OUTPUT_VARIABLE SDK_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT SDK_REV STREQUAL "a1438dff1d38bd9c65dbd693f0e5db4b9ae91779")
    message(FATAL_ERROR "SDK revision differs from dependencies.json; run tools/bootstrap.sh")
  endif()
  include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(sherman_controller LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(controller src/protocol.cpp src/controller.cpp src/sensors.cpp)
target_include_directories(controller PUBLIC include)
target_compile_options(controller PRIVATE -Wall -Wextra -Werror -Wpedantic)
if(SHERMAN_PICO)
  pico_sdk_init()
  execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}
                  OUTPUT_VARIABLE SHERMAN_BUILD_ID OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  execute_process(COMMAND git diff --quiet HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR} RESULT_VARIABLE DIRTY)
  if(DIRTY)
    string(APPEND SHERMAN_BUILD_ID "-dirty")
  endif()
  add_executable(sherman_pico2 src/main.cpp)
  target_compile_definitions(sherman_pico2 PRIVATE SHERMAN_BUILD_ID="${SHERMAN_BUILD_ID}")
  # Keep our HAL strict without imposing ISO-only warnings on SDK internals.
  set_source_files_properties(src/main.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror;-Wpedantic")
  target_include_directories(sherman_pico2 SYSTEM PRIVATE ${PICO_SDK_PATH}/lib/tinyusb/src)
  target_link_libraries(sherman_pico2 controller pico_stdlib pico_rand hardware_i2c hardware_uart hardware_watchdog hardware_sync)
  pico_enable_stdio_uart(sherman_pico2 0)
  pico_enable_stdio_usb(sherman_pico2 1)
  pico_set_program_name(sherman_pico2 "Sherman W1 diagnostic-only")
  pico_set_program_version(sherman_pico2 "0.1.0-${SHERMAN_BUILD_ID}")
  pico_add_extra_outputs(sherman_pico2)
else()
  enable_testing()
  if(SHERMAN_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(controller PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(controller PUBLIC -fsanitize=address,undefined)
  endif()
  foreach(name core sensor)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_compile_options(${name}_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
    target_link_libraries(${name}_test PRIVATE controller)
    add_test(NAME ${name} COMMAND ${name}_test)
  endforeach()
  add_executable(codec_fixture tests/codec_fixture.cpp)
  target_link_libraries(codec_fixture PRIVATE controller)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/contracts_test.py $<TARGET_FILE:codec_fixture>)
endif()
